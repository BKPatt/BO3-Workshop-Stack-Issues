cmake_minimum_required(VERSION 3.20)
project(BO3WorkshopStack LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../workshop/ui/bo3stack/frontend.lua")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../workshop/ui/bo3stack/frontend.lua" STACK_FRONTEND_LUA)
# MSVC limits the size of each literal. Adjacent per-line raw literals retain
# the exact Lua bytes without imposing that limit on the complete UI source.
string(FIND "${STACK_FRONTEND_LUA}" ")stackui\"" STACK_RAW_DELIMITER)
if(NOT STACK_RAW_DELIMITER EQUAL -1)
  message(FATAL_ERROR "Frontend Lua contains the reserved C++ raw-string delimiter")
endif()
string(REPLACE "\n" "\n)stackui\"\nR\"stackui(" STACK_FRONTEND_LUA "${STACK_FRONTEND_LUA}")
configure_file(src/frontend_source.hpp.in generated/frontend_source.hpp @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
add_subdirectory(third_party/minhook)
add_library(bo3_stack SHARED src/extension.cpp)
target_link_libraries(bo3_stack PRIVATE minhook bcrypt user32)
target_compile_definitions(bo3_stack PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(BO3Stack src/bootstrap.cpp)
target_compile_definitions(BO3Stack PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(BO3Stack PRIVATE psapi bcrypt)
include(CTest)
